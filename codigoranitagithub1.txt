# ==== 2) Gráfica de barras – Presente primero, luego Futuro ==== #
# Nacional
areas_nat <- areas %>% filter(Region == "Mexico (national)", Threshold %in% c("700","800"))
areas_nat$Scenario <- factor(areas_nat$Scenario, levels = c("Present","Future_RCP85_2050"))

p_nat <- ggplot(areas_nat, aes(x = Threshold, y = Area_km2, fill = Scenario)) +
  geom_col(position = position_dodge(width = 0.7), width = 0.65, color = "grey20") +

  scale_y_continuous(labels = scales::label_number(big.mark=",")) +
  labs(title = "Suitable area in Mexico (national)",

       x = "Threshold", y = "Area (km²)", fill = "Scenario") +
  theme_minimal(base_size = 12) +
  theme(legend.position = "top")

elev_path   <- "ALTITUDBIENMAXENT.asc"

if(file.exists(elev_path) && file.exists(occ_csv)){
  elev <- terra::rast(elev_path)

  occ_df <- read.csv(occ_csv)
  # Cambia estos nombres si tus campos se llaman distinto (p.ej. "lon","lat")
  lon_col <- "decimalLongitude"
  lat_col <- "decimalLatitude"
  stopifnot(all(c(lon_col, lat_col) %in% names(occ_df)))

  pts <- terra::vect(occ_df, geom = c(lon_col, lat_col), crs = "EPSG:4326")

  # Paleta colorblind-friendly con cafés (Brewer BrBG)
  pal_brbg <- colorRampPalette(RColorBrewer::brewer.pal(11, "BrBG"))(100)

  png("outputs/elevation_occurrences.png", width = 1800, height = 1200, res = 200)
  plot(elev, col = pal_brbg, main = "Occurrences of H. variolosus over a Mexico elevation map")
  points(pts, col = "black", pch = 20, cex = 0.7)
  dev.off()

  pdf("outputs/elevation_occurrences.pdf", width = 9, height = 6)
  plot(elev, col = pal_brbg, main = "Occurrences of H. variolosus over a Mexico elevation map")
  points(pts, col = "black", pch = 20, cex = 0.7)

# ==== 4) Utilidades opcionales para rásteres de idoneidad y umbrales ==== #
# Si cuentas con rásteres continuos (0–1000) para Presente/Futuro, puedes crear mapas binarios
# por umbral y un panel 2x2. Ajusta los nombres de archivos si los tienes.

# Ejemplo de nombres (ajústalos a los tuyos o comenta si no los tienes):
present_raster_path <- "present_suitability.tif"       # 0–1000
future_raster_path  <- "future_suitability_rcp85.tif"  # 0–1000

make_binary_by_threshold <- function(r, thr){
  # r: SpatRaster continuo (0–1000); thr: umbral (700 u 800)
  # Devuelve raster binario (1 = >= thr; NA = < thr)
  x <- r
  x[x < thr] <- NA
  x[x >= thr] <- 1
  return(x)
}

if(file.exists(present_raster_path) && file.exists(future_raster_path)){
  r_p <- rast(present_raster_path)
  r_f <- rast(future_raster_path)

  b_p700 <- make_binary_by_threshold(r_p, 700)
  b_p800 <- make_binary_by_threshold(r_p, 800)
  b_f700 <- make_binary_by_threshold(r_f, 700)
  b_f800 <- make_binary_by_threshold(r_f, 800)

  png("outputs/panel_present_future_thresholds.png", width = 2400, height = 1600, res = 220)
  par(mfrow = c(2,2), mar = c(4,4,3,6))
  plot(b_p700, col = c(NA,"#2C7FB8"), main = "A) Present (≥700)")
  plot(b_p800, col = c(NA,"#253494"), main = "B) Present (≥800)")
  plot(b_f700, col = c(NA,"#E6550D"), main = "C) Future 2050 (≥700)")
  plot(b_f800, col = c(NA,"#A63603"), main = "D) Future 2050 (≥800)")
  dev.off()
}
